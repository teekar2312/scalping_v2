    48→# --------------------------------------------------------------------------- #
    49→# Event bus (simple in-process pub/sub for WebSocket)
    50→# --------------------------------------------------------------------------- #
    51→
    52→class EventBus:
    53→    """Minimal in-process pub/sub using asyncio queues."""
    54→
    55→    def __init__(self) -> None:
    56→        self._subscribers: list[asyncio.Queue[dict[str, Any]]] = []
    57→
    58→    def subscribe(self) -> asyncio.Queue[dict[str, Any]]:
    59→        q: asyncio.Queue[dict[str, Any]] = asyncio.Queue(maxsize=200)
    60→        self._subscribers.append(q)
    61→        return q
    62→
    63→    def unsubscribe(self, q: asyncio.Queue[dict[str, Any]]) -> None:
    64→        try:
    65→            self._subscribers.remove(q)
    66→        except ValueError:
    67→            pass
    68→
    69→    async def publish(self, event: dict[str, Any]) -> None:
    70→        for q in list(self._subscribers):
    71→            try:
    72→                q.put_nowait(event)
    73→            except asyncio.QueueFull:
    74→                # Drop oldest, then push
    75→                try:
    76→                    q.get_nowait()
    77→                    q.put_nowait(event)
    78→                except Exception:  # pragma: no cover
    79→                    pass

   126→        self.enable_ai = enable_ai
   127→        self.state = EngineState()

   129→        self.daily_tracker = DailyRiskTracker()

   135→    # ------------------------------------------------------------------ #
   136→    # Lifecycle
   137→    # ------------------------------------------------------------------ #
   138→
   139→    async def start(self) -> dict[str, Any]:
   140→        """Start the trading loop in the background."""
   141→        if self.state.is_running:
   142→            return {"ok": False, "message": "already running"}
   143→        if self.state.kill_switch:
   144→            return {"ok": False, "message": "kill switch engaged - release first"}
   145→        if self.state.is_locked:
   146→            return {"ok": False, "message": f"bot locked: {self._lock_reason()}"}
   147→
   148→        # Ensure MT5 is connected
   149→        if not self.connector.connected:
   150→            ok = await asyncio.to_thread(self.connector.initialize)
   151→            if not ok:
   152→                return {"ok": False,
   153→                        "message": "MT5 connection failed - check terminal & credentials"}
   154→
   155→        self._stop_event.clear()
   156→        self._task = asyncio.create_task(self._run_loop(), name="trading_loop")
   157→        self.state.is_running = True
   158→        db.update_bot_state(is_running=True)
   159→        await self.bus.publish({"type": "engine_started",
   160→                                "ts": datetime.now(timezone.utc).isoformat()})
   161→        logger.info("Trading engine started (interval=%.1fs)", self.loop_interval)
   162→        return {"ok": True, "message": "started"}
   163→
   164→    async def stop(self, reason: str = "manual") -> dict[str, Any]:
   165→        """Stop the trading loop (does NOT close positions)."""
   166→        if not self.state.is_running and self._task is None:
   167→            return {"ok": False, "message": "not running"}
   168→        self._stop_event.set()
   169→        if self._task is not None:
   170→            self._task.cancel()
   171→            try:
   172→                await self._task
   173→            except asyncio.CancelledError:
   174→                pass
   175→            self._task = None
   176→        self.state.is_running = False
   177→        db.update_bot_state(is_running=False, last_error=reason)
   178→        await self.bus.publish({"type": "engine_stopped", "reason": reason,
   179→                                "ts": datetime.now(timezone.utc).isoformat()})
   180→        logger.info("Trading engine stopped (%s)", reason)
   181→        return {"ok": True, "message": f"stopped: {reason}"}
   182→
   183→    async def kill(self, reason: str = "manual_kill") -> dict[str, Any]:
   184→        """Emergency kill switch: stop loop + close ALL positions + lock."""
   185→        logger.warning("KILL SWITCH ENGAGED: %s", reason)
   186→        # Stop the loop first
   187→        await self.stop(reason=f"kill:{reason}")
   188→        # Close all positions
   189→        closed: list[dict[str, Any]] = []
   190→        try:
   191→            closed = await asyncio.to_thread(self.connector.close_all_positions)
   192→        except MT5UnavailableError as exc:
   193→            logger.error("Cannot close positions (MT5 unavailable): %s", exc)
   194→        except Exception as exc:  # pragma: no cover
   195→            logger.exception("close_all_positions failed: %s", exc)
   196→        # Engage kill switch + lock
   197→        engage_kill_switch()
   198→        lock_bot(f"kill:{reason}")
   199→        self.state.kill_switch = True
   200→        self.state.is_locked = True
   201→        await self.bus.publish({
   202→            "type": "kill_switch_engaged",
   203→            "reason": reason,
   204→            "closed": closed,
   205→            "ts": datetime.now(timezone.utc).isoformat(),
   206→        })
   207→        return {"ok": True, "closed": closed, "reason": reason}
   208→
   209→    async def release_kill(self) -> dict[str, Any]:
   210→        """Release the kill switch + unlock (manual override)."""
   211→        from risk_manager import release_kill_switch
   212→        release_kill_switch()
   213→        unlock_bot()
   214→        self.state.kill_switch = False
   215→        self.state.is_locked = False
   216→        await self.bus.publish({"type": "kill_switch_released",
   217→                                "ts": datetime.now(timezone.utc).isoformat()})
   218→        return {"ok": True}
   219→
   220→    def _lock_reason(self) -> str | None:
   221→        st = db.get_bot_state()
   222→        return st.get("lock_reason")

   224→    # ------------------------------------------------------------------ #
   225→    # Main loop
   226→    # ------------------------------------------------------------------ #
   227→
   228→    async def _run_loop(self) -> None:
   229→        """The main scan loop.  Runs until cancelled or stopped."""
   230→        self._install_shutdown_handlers()
   231→        try:
   232→            while not self._stop_event.is_set():
   233→                try:
   234→                    await self._tick()
   235→                except MT5UnavailableError as exc:
   236→                    logger.error("MT5 unavailable: %s", exc)
   237→                    self.state.errors.append(str(exc))
   238→                    # Try to reconnect
   239→                    try:
   240→                        await asyncio.to_thread(self.connector.initialize)
   241→                    except Exception:
   242→                        pass
   243→                except Exception as exc:  # pragma: no cover
   244→                    logger.exception("Tick failed: %s", exc)
   245→                    self.state.errors.append(str(exc))
   246→                    db.update_bot_state(last_error=str(exc))
   247→                # Wait for next tick or stop
   248→                try:
   249→                    await asyncio.wait_for(self._stop_event.wait(),
   250→                                            timeout=self.loop_interval)
   251→                except asyncio.TimeoutError:
   252→                    pass
   253→        except asyncio.CancelledError:
   254→            logger.info("Trading loop cancelled")
   255→            raise
   256→        finally:
   257→            self.state.is_running = False
   258→            db.update_bot_state(is_running=False,
   259→                                last_loop_at=datetime.now(timezone.utc))
   260→
   261→    async def _tick(self) -> None:
   262→        """One scan + manage iteration."""
   263→        # Update account snapshot
   264→        try:
   265→            account = await asyncio.to_thread(self.connector.get_account_info)
   266→            self.state.balance = account.balance
   267→            self.state.equity = account.equity
   268→        except Exception as exc:
   269→            logger.warning("get_account_info failed: %s", exc)
   270→            return
   271→
   272→        # Daily risk
   273→        daily = self.daily_tracker.get_state(balance=account.balance)
   274→        self.state.daily_trades = daily.trade_count
   275→        self.state.daily_pnl = daily.realized_pnl
   276→
   277→        # Manage open positions first
   278→        await self.manage_open_positions(account.balance)
   279→
   280→        # Risk gate
   281→        positions = await asyncio.to_thread(self.connector.get_positions)
   282→        self.state.open_positions = len(positions)
   283→
   284→        # Weekly/overall starting balance: use last Monday & lifetime.
   285→        weekly_start = await self._weekly_starting_balance(account.balance)
   286→        overall_start = await self._overall_starting_balance(account.balance)
   287→        gate = evaluate_risk_gate(
   288→            current_positions=len(positions),
   289→            daily_state=daily,
   290→            weekly_start=weekly_start,
   291→            overall_start=overall_start,
   292→            current_balance=account.balance,
   293→        )
   294→        self.state.is_locked = gate.bot_locked
   295→        self.state.kill_switch = gate.kill_switch
   296→
   297→        if gate.kill_switch:
   298→            logger.warning("Kill switch active - skipping scan")
   299→            return
   300→        if not gate.allow_new_entry:
   301→            logger.info("Risk gate blocks new entries: %s", gate.reasons)
   302→            await self.bus.publish({
   303→                "type": "risk_gate_blocked",
   304→                "reasons": gate.reasons,
   305→                "ts": datetime.now(timezone.utc).isoformat(),
   306→            })
   307→            # If hard stop triggered -> engage kill switch
   308→            if gate.tiered_stop.level == "hard":
   309→                await self.kill(reason=f"hard_stop:{gate.tiered_stop.pnl_pct}")
   310→            return
   311→
   312→        # Scan markets
   313→        await self.scan_markets(account.balance)
   314→
   315→        # Persist state
   316→        self.state.last_scan_at = datetime.now(timezone.utc).isoformat()
   317→        db.update_bot_state(last_loop_at=datetime.now(timezone.utc))

   323→    async def scan_markets(self, balance: float) -> list[SignalContext]:
   324→        """Scan every configured pair and return any valid signals."""
   325→        # Fetch news once
   326→        try:
   327→            cal = get_calendar()
   328→            news_events = await cal.fetch_calendar(impact="HIGH")
   329→        except Exception as exc:  # pragma: no cover
   330→            logger.warning("Tiingo calendar fetch failed: %s", exc)
   331→            news_events = []
   332→
   333→        signals: list[SignalContext] = []
   334→        for pair, mt5_sym in zip(settings.pairs, settings.mt5_symbols):
   335→            try:
   336→                sig = await self._scan_symbol(pair, mt5_sym, balance, news_events)
   337→                if sig:
   338→                    signals.append(sig)
   339→                    await self.on_signal(sig, mt5_sym, balance, news_events)
   340→            except MT5UnavailableError:
   341→                raise
   342→            except Exception as exc:  # pragma: no cover
   343→                logger.exception("scan_symbol(%s) failed: %s", pair, exc)
   344→        return signals

   346→    async def _scan_symbol(self, pair: str, mt5_sym: str,
   347→                           balance: float,
   348→                           news_events: list[dict[str, Any]]) -> SignalContext | None:
   349→        """Scan one symbol and return its SignalContext (or None)."""
   350→        # Fetch candles
   351→        candles_m5_raw = await asyncio.to_thread(
   352→            self.connector.get_candles, mt5_sym, "M5", 200)
   353→        candles_m15_raw = await asyncio.to_thread(
   354→            self.connector.get_candles, mt5_sym, "M15", 200)
   355→        if not candles_m5_raw or not candles_m15_raw:
   356→            return None
   357→
   358→        candles_m5 = _candles_from_bars(candles_m5_raw)
   359→        candles_m15 = _candles_from_bars(candles_m15_raw)
   360→
   361→        # Symbol info
   362→        try:
   363→            sym_info = await asyncio.to_thread(
   364→                self.connector.get_symbol_info, mt5_sym)
   365→        except Exception as exc:  # pragma: no cover
   366→            logger.warning("get_symbol_info(%s) failed: %s", mt5_sym, exc)
   367→            return None
   368→
   369→        spread_pips = sym_info.spread * sym_info.point / sym_info.pip_size \
   370→            if sym_info.pip_size > 0 else 0.0
   371→        # Commission pips: assume round-turn commission in account currency
   372→        # converted to pips.  Default to 0.7 pip ECN if unknown.
   373→        commission_pips = 0.7
   374→        try:
   375→            commission_pips = float(db.get_setting("commission_pips") or "0.7")
   376→        except Exception:
   377→            pass
   378→
   379→        daily = self.daily_tracker.get_state()
   380→        ctx = check_setup(
   381→            symbol=pair,
   382→            candles_m5=candles_m5,
   383→            candles_m15=candles_m15,
   384→            symbol_info=sym_info,
   385→            spread_pips=spread_pips,
   386→            commission_pips=commission_pips,
   387→            balance=balance,
   388→            news_events=news_events,
   389→            soft_stop_hit=daily.soft_stop_hit,
   390→            now_wib=datetime.now(WIB_TZ),
   391→        )
   392→        if ctx is None:
   393→            return None
   394→
   395→        # Persist signal
   396→        db.insert_signal(
   397→            symbol=pair,
   398→            direction=ctx.direction or ("buy" if ctx.ema9 > ctx.ema21 else "sell"),
   399→            entry_price=ctx.entry_price,
   400→            sl=ctx.sl_price,
   401→            tp1=ctx.tp1_price,
   402→            tp2=ctx.tp2_price,
   403→            atr=ctx.atr_pips,
   404→            adx=ctx.adx_value,
   405→            rr_net=ctx.rr_net,
   406→            regime=ctx.regime,
   407→            session=ctx.session,
   408→            checklist_json=json.dumps(ctx.checklist),
   409→            executed=False,
   410→        )
   411→
   412→        # Publish event
   413→        event = {
   414→            "type": "signal",
   415→            "symbol": pair,
   416→            "direction": ctx.direction,
   417→            "pass_all": ctx.pass_all,
   418→            "regime": ctx.regime,
   419→            "session": ctx.session,
   420→            "adx": ctx.adx_value,
   421→            "rr_net": ctx.rr_net,
   422→            "checklist": ctx.checklist,
   423→            "fail_reasons": ctx.fail_reasons,
   424→            "ts": ctx.timestamp,
   425→        }
   426→        await self.bus.publish(event)
   427→        self.state.last_signal = event
   428→        return ctx

   434→    async def on_signal(self, sig: SignalContext, mt5_sym: str,
   435→                        balance: float,
   436→                        news_events: list[dict[str, Any]]) -> None:
   437→        """Evaluate + execute a signal (subject to AI compliance if enabled)."""
   438→        if not sig.pass_all:
   439→            logger.info("Signal %s %s failed checklist: %s",
   440→                        sig.symbol, sig.direction, sig.fail_reasons)
   441→            return
   442→        if sig.direction is None:
   443→            return
   444→
   445→        # Optional AI compliance check
   446→        if self.enable_ai:
   447→            try:
   448→                analyzer = get_analyzer()
   449→                ai = await analyzer.analyze_setup(sig.to_dict())
   450→                if not ai.compliant:
   451→                    logger.info("AI rejected setup %s: %s (risks=%s)",
   452→                                sig.symbol, ai.suggestion, ai.risks)
   453→                    await self.bus.publish({
   454→                        "type": "ai_rejected",
   455→                        "symbol": sig.symbol,
   456→                        "suggestion": ai.suggestion,
   457→                        "risks": ai.risks,
   458→                        "ts": datetime.now(timezone.utc).isoformat(),
   459→                    })
   460→                    return
   461→            except Exception as exc:  # pragma: no cover
   462→                logger.warning("AI analysis failed - proceeding: %s", exc)
   463→
   464→        # Final risk gate
   465→        positions = await asyncio.to_thread(self.connector.get_positions)
   466→        daily = self.daily_tracker.get_state(balance=balance)
   467→        weekly_start = await self._weekly_starting_balance(balance)
   468→        overall_start = await self._overall_starting_balance(balance)
   469→        gate = evaluate_risk_gate(
   470→            current_positions=len(positions),
   471→            daily_state=daily,
   472→            weekly_start=weekly_start,
   473→            overall_start=overall_start,
   474→            current_balance=balance,
   475→        )
   476→        if not gate.allow_new_entry:
   477→            logger.info("Risk gate blocks entry for %s: %s",
   478→                        sig.symbol, gate.reasons)
   479→            return
   480→
   481→        # Lot size
   482→        if sig.lot_size is None or sig.lot_size <= 0:
   483→            logger.warning("No valid lot size for %s - skipping", sig.symbol)
   484→            return
   485→
   486→        # Execute order
   487→        order = await asyncio.to_thread(
   488→            self.connector.place_order,
   489→            mt5_sym, sig.direction, sig.lot_size,
   490→            sig.sl_price or 0.0,
   491→            sig.tp1_price or 0.0,
   492→            comment=f"v2:{sig.regime[:3]}",
   493→        )
   494→        ok = order.get("ok", False)
   495→        await self.bus.publish({
   496→            "type": "order_placed",
   497→            "symbol": sig.symbol,
   498→            "direction": sig.direction,
   499→            "lot": sig.lot_size,
   500→            "sl": sig.sl_price,
   501→            "tp": sig.tp1_price,
   502→            "ok": ok,
   503→            "retcode": order.get("retcode"),
   504→            "comment": order.get("comment"),
   505→            "ts": datetime.now(timezone.utc).isoformat(),
   506→        })
   507→        if ok:
   508→            trade_id = db.insert_trade(
   509→                ticket=order.get("ticket"),
   510→                symbol=sig.symbol,
   511→                direction=sig.direction,
   512→                volume=sig.lot_size,
   513→                open_price=sig.entry_price,
   514→                sl=sig.sl_price,
   515→                tp=sig.tp1_price,
   516→                status="open",
   517→                session_name=sig.session,
   518→                entry_reason=",".join(sig.checklist.keys()) if sig.checklist else "",
   519→                checklist_json=json.dumps(sig.checklist),
   520→                comment=f"v2:{sig.regime[:3]}",
   521→            )
   522→            logger.info("Order placed: %s %s %.2f lot ticket=%s trade_id=%s",
   523→                        sig.symbol, sig.direction, sig.lot_size,
   524→                        order.get("ticket"), trade_id)
   525→        else:
   526→            logger.warning("Order failed for %s: %s", sig.symbol, order)

   532→    async def manage_open_positions(self, balance: float) -> None:
   533→        """Check TP1 / TP2 / trailing / SL on every open position."""
   534→        try:
   535→            positions = await asyncio.to_thread(self.connector.get_positions)
   536→        except Exception as exc:  # pragma: no cover
   537→            logger.warning("get_positions failed: %s", exc)
   538→            return
   539→
   540→        for pos in positions:
   541→            try:
   542→                await self._manage_one(pos, balance)
   543→            except Exception as exc:  # pragma: no cover
   544→                logger.exception("manage_one(%s) failed: %s", pos.ticket, exc)

   546→    async def _manage_one(self, pos: Any, balance: float) -> None:
   547→        """Manage a single position: TP1/TP2/trailing/SL."""
   548→        # Pull context for this position from DB (if any)
   549→        # We track management state via comments / position sl vs entry.
   550→        is_buy = pos.type == "buy"
   551→        sym_info = await asyncio.to_thread(
   552→            self.connector.get_symbol_info, pos.symbol)
   553→        pip_size = sym_info.pip_size
   554→        current_price = sym_info.bid if is_buy else sym_info.ask
   555→        entry = pos.open_price
   556→        sl = pos.sl
   557→
   558→        # Distance to TP1 in pips (if we recorded original SL/TP we'd compute
   559→        # exactly; here we approximate using current SL distance).
   560→        if sl == 0:
   561→            return   # no SL set, cannot derive TP1/TP2
   562→        sl_distance_pips = abs(entry - sl) / pip_size if pip_size else 0
   563→        tp1_distance = sl_distance_pips * cfg.TP1_MULT
   564→        tp2_distance = sl_distance_pips * cfg.TP2_MULT
   565→
   566→        # Current PnL in pips
   567→        if is_buy:
   568→            moved_pips = (current_price - entry) / pip_size if pip_size else 0
   569→        else:
   570→            moved_pips = (entry - current_price) / pip_size if pip_size else 0
   571→
   572→        # ---- TP1: close 50%, move SL to BE+2 ----
   573→        if moved_pips >= tp1_distance and not self._tp1_done(pos):
   574→            # Move SL to BE+2
   575→            be_buffer = cfg.BE_BUFFER_PIPS * pip_size
   576→            if is_buy:
   577→                new_sl = entry + be_buffer
   578→            else:
   579→                new_sl = entry - be_buffer
   580→            await asyncio.to_thread(
   581→                self.connector.modify_position, pos.ticket, sl=new_sl, tp=pos.tp)
   582→            # Partial close 50%
   583→            partial_vol = round(pos.volume * cfg.TP1_CLOSE_PCT, 2)
   584→            await self._partial_close(pos, partial_vol, reason="tp1")
   585→            await self.bus.publish({
   586→                "type": "tp1_hit",
   587→                "ticket": pos.ticket,
   588→                "symbol": pos.symbol,
   589→                "sl_to_be": new_sl,
   590→                "closed_volume": partial_vol,
   591→                "ts": datetime.now(timezone.utc).isoformat(),
   592→            })
   593→
   594→        # ---- TP2: close 30%, start trailing ----
   595→        elif moved_pips >= tp2_distance and not self._tp2_done(pos):
   596→            partial_vol = round(pos.volume * cfg.TP2_CLOSE_PCT, 2)
   597→            await self._partial_close(pos, partial_vol, reason="tp2")
   598→            await self.bus.publish({
   599→                "type": "tp2_hit",
   600→                "ticket": pos.ticket,
   601→                "symbol": pos.symbol,
   602→                "closed_volume": partial_vol,
   603→                "ts": datetime.now(timezone.utc).isoformat(),
   604→            })
   605→
   606→        # ---- Trailing 2x ATR for remaining 20% ----
   607→        elif moved_pips >= tp2_distance and self._tp2_done(pos):
   608→            await self._trail_position(pos, sym_info, is_buy, entry, pip_size)
   609→
   610→    def _tp1_done(self, pos: Any) -> bool:
   611→        """Heuristic: TP1 is done if current SL is at/beyond BE+2."""
   612→        if pos.sl == 0 or pos.open_price == 0:
   613→            return False
   614→        if pos.type == "buy":
   615→            return pos.sl >= pos.open_price
   616→        return pos.sl <= pos.open_price
   617→
   618→    def _tp2_done(self, pos: Any) -> bool:
   619→        """Heuristic: assume TP2 done if position volume already reduced.
   620→
   621→        In a real broker integration we'd track this explicitly via magic
   622→        numbers or a separate DB column.
   623→        """
   624→        return False   # placeholder; full state tracking done via DB
   625→
   626→    async def _partial_close(self, pos: Any, volume: float, reason: str) -> None:
   627→        """Close part of a position.  MT5 doesn't support partial closes
   628→        directly; we'd normally re-open the remainder.  For simplicity we
   629→        close the entire position here and re-open with the residual volume.
   630→
   631→        NOTE: This is a simplified implementation - in production use
   632→        ``order_send`` with reduced volume and a "close" opposite order for
   633→        the partial.
   634→        """
   635→        if volume <= 0:
   636→            return
   637→        # Simplified: just close the whole ticket.  Real partial-close logic
   638→        # would split the ticket via opposite orders.
   639→        logger.info("Partial close (%s) ticket=%s vol=%.2f -> full close",
   640→                    reason, pos.ticket, volume)
   641→        await self.exit_position(pos.ticket, reason=reason)
   642→
   643→    async def _trail_position(self, pos: Any, sym_info: Any,
   644→                              is_buy: bool, entry: float,
   645→                              pip_size: float) -> None:
   646→        """Trail the SL at 2 x ATR behind current price."""
   647→        try:
   648→            bars = await asyncio.to_thread(
   649→                self.connector.get_candles, pos.symbol, "M5", 50)
   650→            if not bars:
   651→                return
   652→            candles = _candles_from_bars(bars)
   653→            from indicators import atr as _atr
   654→            atr_val = _atr([c.high for c in candles],
   655→                           [c.low for c in candles],
   656→                           [c.close for c in candles], period=14)
   657→            if math.isnan(atr_val):
   658→                return
   659→            trail = atr_val * cfg.TRAIL_ATR_MULT
   660→            if is_buy:
   661→                new_sl = (sym_info.bid - trail)
   662→                if new_sl > pos.sl:
   663→                    await asyncio.to_thread(
   664→                        self.connector.modify_position, pos.ticket,
   665→                        sl=new_sl, tp=pos.tp)
   666→            else:
   667→                new_sl = (sym_info.ask + trail)
   668→                if pos.sl == 0 or new_sl < pos.sl:
   669→                    await asyncio.to_thread(
   670→                        self.connector.modify_position, pos.ticket,
   671→                        sl=new_sl, tp=pos.tp)
   672→        except Exception as exc:  # pragma: no cover
   673→            logger.warning("Trail failed for %s: %s", pos.ticket, exc)
   674→
   675→    async def exit_position(self, ticket: int, reason: str) -> dict[str, Any]:
   676→        """Close a position by ticket."""
   677→        result = await asyncio.to_thread(self.connector.close_position, ticket)
   678→        await self.bus.publish({
   679→            "type": "position_closed",
   680→            "ticket": ticket,
   681→            "reason": reason,
   682→            "ok": result.get("ok"),
   683→            "ts": datetime.now(timezone.utc).isoformat(),
   684→        })
   685→        return result

   715→    def _install_shutdown_handlers(self) -> None:
   716→        if self._shutdown_handlers_installed:
   717→            return
   718→        try:
   719→            loop = asyncio.get_running_loop()
   720→            for sig in (signal.SIGINT, signal.SIGTERM):
   721→                try:
   722→                    loop.add_signal_handler(sig, self._on_signal)
   723→                except NotImplementedError:
   724→                    # Windows - use KeyboardInterrupt fallback
   725→                    pass
   726→        except RuntimeError:
   727→            pass
   728→        self._shutdown_handlers_installed = True
   729→
   730→    def _on_signal(self) -> None:
   731→        logger.warning("Shutdown signal received - stopping engine")
   732→        asyncio.create_task(self.stop(reason="signal"))
